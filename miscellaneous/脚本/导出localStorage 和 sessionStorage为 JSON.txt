(function exportWebStorage() {
  const data = {
    localStorage: {},
    sessionStorage: {}
  };

  for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);
    data.localStorage[key] = localStorage.getItem(key);
  }

  for (let i = 0; i < sessionStorage.length; i++) {
    const key = sessionStorage.key(i);
    data.sessionStorage[key] = sessionStorage.getItem(key);
  }

  const blob = new Blob([JSON.stringify(data, null, 2)], {
    type: 'application/json'
  });

  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = `${location.hostname}-storage-${new Date()
    .toISOString()
    .slice(0, 19)
    .replace(/[:T]/g, '-')}.json`;

  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(a.href);

  console.log('导出完成：', data);
})();

(function importWebStorage() {
  const input = document.createElement('input');
  input.type = 'file';
  input.accept = '.json,application/json';
  input.style.display = 'none';
  document.body.appendChild(input);

  input.onchange = async (e) => {
    document.body.removeChild(input);

    const file = e.target.files[0];
    if (!file) return;

    let data;
    try {
      data = JSON.parse(await file.text());
    } catch (err) {
      alert('JSON 解析失败：' + err.message);
      return;
    }

    // 兼容两种格式：
    // 1) { localStorage: {...}, sessionStorage: {...} }
    // 2) 直接就是 localStorage 的键值对象
    const hasNested =
      data &&
      typeof data === 'object' &&
      data.localStorage &&
      typeof data.localStorage === 'object';

    const localData = hasNested ? data.localStorage : data;
    const sessionData =
      hasNested && data.sessionStorage && typeof data.sessionStorage === 'object'
        ? data.sessionStorage
        : {};

    const localKeys = Object.keys(localData);
    const sessionKeys = Object.keys(sessionData);

    if (!localKeys.length && !sessionKeys.length) {
      alert('没有找到可导入的数据。');
      return;
    }

    if (
      !confirm(
        `将导入 ${localKeys.length} 个 localStorage 项，${sessionKeys.length} 个 sessionStorage 项，覆盖同名键。继续？`
      )
    ) {
      return;
    }

    for (const [key, value] of Object.entries(localData)) {
      localStorage.setItem(
        key,
        typeof value === 'string' ? value : JSON.stringify(value)
      );
    }

    for (const [key, value] of Object.entries(sessionData)) {
      sessionStorage.setItem(
        key,
        typeof value === 'string' ? value : JSON.stringify(value)
      );
    }

    alert('导入完成，刷新页面后生效。');
    location.reload();
  };

  input.click();
})();
